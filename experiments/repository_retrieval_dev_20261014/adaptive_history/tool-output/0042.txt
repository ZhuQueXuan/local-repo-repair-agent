
class JsonPickleChecks(unittest.TestCase):
    def assert_roundtrip(self, original, protocol):
        restored = pickle.loads(pickle.dumps(original, protocol=protocol))
        self.assertIs(type(restored), type(original))
        for field in ["msg", "doc", "pos", "lineno", "colno", "args"]:
            self.assertEqual(getattr(restored, field), getattr(original, field))
        self.assertEqual(str(restored), str(original))

    def test_direct_exception_all_pickle_protocols(self):
        for protocol in range(pickle.HIGHEST_PROTOCOL + 1):
            with self.subTest(protocol=protocol):
                self.assert_roundtrip(requests.exceptions.JSONDecodeError("Extra data", '{"a": 1}{"b": 2}', 8), protocol)

    def test_response_json_exception_roundtrip(self):
        for content, encoding in [(b'{"a":}', "utf-8"), ('{"文字":}'.encode("utf-8"), "utf-8"),
                                  ('{"a":}'.encode("utf-16"), None)]:
            with self.subTest(content=content, encoding=encoding):
                try:
                    response_with(content, encoding).json()
                except requests.exceptions.JSONDecodeError as error:
                    self.assert_roundtrip(error, pickle.HIGHEST_PROTOCOL)
                else:
                    self.fail("Invalid JSON unexpectedly decoded")

    def test_multiline_error_location_survives(self):
        self.assert_roundtrip(requests.exceptions.JSONDecodeError("Expected value", '{\n  "value":\n}', 13), 2)

    def test_invalid_response_raises_public_json_exception(self):
        with self.assertRaises(requests.exceptions.JSONDecodeError) as caught:
            response_with(b'{"bad":}').json()
        self.assertIsInstance(caught.exception, requests.exceptions.RequestException)
        self.assertIsInstance(caught.exception, ValueError)
        self.assertEqual(caught.exception.doc, '{"bad":}')

    def test_valid_response_is_unchanged(self):
        self.assertEqual(response_with(b'{"items": [1, null, true]}').json(), {"items": [1, None, True]})

    def test_other_request_exceptions_still_roundtrip(self):
        for kind in [requests.exceptions.RequestException, requests.exceptions.Timeout, requests.exceptions.HTTPError]:
            with self.subTest(kind=kind.__name__):
                error = kind("ordinary failure")
                restored = pickle.loads(pickle.dumps(error))
                self.assertIs(type(restored), kind)
                self.assertEqual(restored.args, error.args)
