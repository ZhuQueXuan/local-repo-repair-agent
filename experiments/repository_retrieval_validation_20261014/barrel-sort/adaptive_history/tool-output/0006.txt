     2	        if index < 0:
     3	            index += len(self)
     4	        rel_idx, lists = index, self.lists
     5	        for list_idx in range(len(lists)):
     6	            len_list = len(lists[list_idx])
     7	            if rel_idx < len_list:
     8	                break
     9	            rel_idx -= len_list
    10	        if rel_idx < 0:
    11	            return None, None
    12	        return list_idx, rel_idx

    14	    def _balance_list(self, list_idx):
    15	        if list_idx < 0:
    16	            list_idx += len(self.lists)
    17	        cur_list, len_self = self.lists[list_idx], len(self)
    18	        size_limit = self._cur_size_limit
    19	        if len(cur_list) > size_limit:
    20	            half_limit = size_limit // 2
    21	            while len(cur_list) > half_limit:
    22	                next_list_idx = list_idx + 1
    23	                self.lists.insert(next_list_idx, cur_list[-half_limit:])
    24	                del cur_list[-half_limit:]

    26	        return False

    83	        if step is not None and abs(step) > 1:  # punt

    97	        if stop_list_idx is None:
