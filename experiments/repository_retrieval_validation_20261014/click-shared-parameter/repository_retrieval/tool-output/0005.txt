     1	        attr_params = attrs.pop("params", None)
     2	        params = attr_params if attr_params is not None else []
     3	
     4	        try:
     5	            decorator_params = f.__click_params__  # type: ignore
     6	        except AttributeError:
     7	            pass
     8	        else:
     9	            del f.__click_params__  # type: ignore
    10	            params.extend(reversed(decorator_params))
    11	
    12	        if attrs.get("help") is None:
    13	            attrs["help"] = f.__doc__
    14	
    15	        cmd = cls(  # type: ignore[misc]
    16	            name=name or f.__name__.lower().replace("_", "-"),  # type: ignore[arg-type]
    17	            callback=f,
    18	            params=params,
    19	            **attrs,
    20	        )
    21	        cmd.__doc__ = f.__doc__
    22	        return cmd
    23	
    24	    if func is not None:
    25	        return decorator(func)
    26	
    27	    return decorator

    48	    """Creates a new :class:`Group` with a function as callback.  This

    75	def argument(*param_decls: str, **attrs: t.Any) -> t.Callable[[FC], FC]:
    76	    """Attaches an argument to the command.  All positional arguments are
    77	    passed as parameter declarations to :class:`Argument`; all keyword
    78	    arguments are forwarded unchanged (except ``cls``).
    79	    This is equivalent to creating an :class:`Argument` instance manually
    80	    and attaching it to the :attr:`Command.params` list.
    81	
    82	    :param cls: the argument class to instantiate.  This defaults to
    83	                :class:`Argument`.
    84	    """
    85	
    86	    def decorator(f: FC) -> FC:
    87	        ArgumentClass = attrs.pop("cls", None) or Argument
    88	        _param_memo(f, ArgumentClass(param_decls, **attrs))
    89	        return f
    90	
    91	    return decorator
