    31	                file_perms = stat.S_IMODE(stat_res.st_mode)

    33	                # default if no destination file exists
    34	                file_perms = self._default_file_perms
    35	                do_chmod = False  # respect the umask
    36

    38	        set_cloexec(fd)
    39	        self.part_file = os.fdopen(fd, self.mode, self.buffering)
    40	
    41	        # if default perms are overridden by the user or previous dest_path
    42	        # chmod away the effects of the umask
    43	        if do_chmod:

    45	                os.chmod(self.part_path, file_perms)
